# map을 이용한 백터 다루기
```c++
struct Nature {
	std::string name;
	std::string upStat;   // 1.1배 되는 스탯
	std::string downStat; // 0.9배 되는 스탯
}; // 포켓몬 성격 map의 성질을 정해줌.

static std::map<std::string, Nature> natureChart = { // std::map< 열쇠타입, 내용물타입 > 변수이름;
	{"Hardy",   {"Hardy",   "", ""}},          // 변동 없음
	{"Lonely",  {"Lonely",  "attack", "defense"}}, // attack 상승 defense 하락을 위한 
	{"Brave",   {"Brave",   "attack", "speed"}},
	{"Adamant", {"Adamant", "attack", "specialAttack"}},
}; // map에 포켓몬이 사용할 성격 집어넣어줌 해당기능은 포켓몬마다 넣으면 메모리 누수이니 static으로 저장

```

# map 의 백터
```c++

string getRandom(){ // 성격을 정해주는 랜덤씨드를 만들기위함.
static std::vector<std::string> natureNames;

for (auto const& pair : natureChart) { // natureChart는 map의 이름 pare은 map의 키.
	natureNames.push_back(pair.first); // pair.first가 key(성격 이름)입니다.
}
int randomIndex = rand() % natureNames.size(); // cmath의 rand()함수를 사용해 랜덤값, 랜덤한index를 만듬
	return natureNames[randomIndex];
}

// natureNames.push_back(pair.first); vetor의 push_vack(), map의 .first()사용
// 이외에도 erase(); begin(); 등 많다. 총 망라할 필요 있어보인다.
```

# vector 기능
```c++
// 반복자 (Iterators)

v.begin()	// 벡터의 첫 번째 원소를 가리키는 반복자 반환
v.end()	// 벡터의 마지막 원소 다음 위치를 가리키는 반복자 반환
v.rbegin()	// 역순(Reverse)으로 첫 번째 원소를 가리킴
v.rend()	// 역순으로 마지막 원소 다음 위치를 가리킴

// 데이터 추가 및 삭제 (Modifiers)

v.push_back(val)	// 벡터의 맨 뒤에 값 추가
v.pop_back()	// 벡터의 맨 뒤 값 삭제
v.insert(pos, val)	// pos 위치에 값 삽입 (반복자 사용)
v.erase(pos)	// pos 위치의 값 삭제
v.erase(start, end)	// 특정 범위의 원소들을 한꺼번에 삭제
v.clear()	// 모든 원소 삭제 (크기는 0이 됨)
v.emplace_back(args) //	객체를 생성해서 맨 뒤에 넣음 (push_back보다 효율적일 때가 많음)

// 데이터 접근 (Element Access)

v[i]	// i번째 원소 접근 (속도가 빠름)
v.at(i)	// i번째 원소 접근 (범위 검사를 함, 안전함)
v.front()	// 첫 번째 원소 참조
v.back()	// 마지막 원소 참조

// 크기 및 용량 (Capacity)

v.size()	// 벡터에 담긴 실제 원소 개수
v.empty()	// 비어있으면 true, 아니면 false
v.capacity()	// 할당된 메모리 공간의 크기 (size와 다를 수 있음)
v.reserve(n)	// 미리 n개의 공간을 예약 (성능 최적화용)
v.resize(n)	// 벡터의 크기를 n으로 강제 조정
```
