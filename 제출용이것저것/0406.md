```c++
struct Nature {
	string name;
	string upStat;   // 1.1배 되는 스탯
	string downStat; // 0.9배 되는 스탯
};
 static map<string, Nature> natureChart = {
	{"Hardy",   {"Hardy",   "", ""}},          // 변동 없음
	{"Lonely",  {"Lonely",  "attack", "defense"}},
	{"Brave",   {"Brave",   "attack", "speed"}},
	{"Adamant", {"Adamant", "attack", "specialAttack"}},
	{"Naughty", {"Naughty", "attack", "specialDefense"}},
	{"Bold",    {"Bold",    "defense", "attack"}},
	{"Docile",  {"Docile",  "", ""}}          // 변동 없음
};

static map<string, Nature> natureChart = {"Hardy"} // 1. 키 값을 지정해준다.
//
static map<string, Nature> natureChart = {"Hardy",{}} // 2 . 밸류 값을 지정해줄 공간을 확보한다.
//
static map<string, Nature> natureChart = {"Hardy", {"Hardy", "", ""}} // 3. 구조체의 요소에 맞춰서 넣고싶은 밸류를 넣는다.
```
