```c++

#include <string>

int intger(int a) {
    return a;
}

{
    std::string str = std::to_string(intger(5));

    if (str != "5") {
        std::cout << "str 은 일단 문자열 5 는 아님!" << std::endl;
    }
    else {
        std::cout << "str 은 문자열 5임!" << std::endl;
    }

    std::cout << (float)intger(5) / intger(2) << std::endl;

    return 0;
```

# do-while문
```c++
int main()
{
    int two = 2;

        std::cout << "세번 보면 살짝 눈이 시큰거려서 눈을 깜박이게 하는 출력문" << std::endl;
```

# 비트 연산자
```c++
int main()
{
	int num1 = 11; 
	int num2 = 9;
	int num3 = 5;
	std::cout << (num1 ^ num2 ^ num3) << std::endl; // 1011 ^ 1001 / 0010 0101/ 0111 == 다른놈 1
	std::cout << (num1 | num2) << std::endl; // 1011 | 1001 1011 == 1인 놈 1
	std::cout << (num1 & num2) << std::endl; // 1011 & 1001 1001 == 같은놈 1
	std::cout << (~num1) << std::endl; // 역산 1111 1111 1111 1111 0100 = -12 // 다 1이면 -1이다.
	std::cout << (~num2) << std::endl; // 1111 1111 1111 1111 0110 = -10
	std::cout << (num1 << 2) << std::endl; // 1011 2칸씩 left shift num1 = 0010 1100 32+8+3 = 44 =11*2*2 
	std::cout << (num1 >> 2) << std::endl; // 1011 2칸씩 right shift num1 == 0010 11(1이하수는 0처리) 2 11/2/2
    
    return 0;
}


```

```c++
//출력문
7
11
9
-12
-10
44
2
```
